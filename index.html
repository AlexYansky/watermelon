function updateWeightHint() { const hintDiv = document.getElementById("weight-hint-text"); hideResult(); const volume = calculateVolume(); if (!Number.isFinite(volume) || volume <= 0) { hintDiv.style.display = "none"; hintDiv.innerHTML = ""; return; } const watermelonType = getSelectedType(); const scoreWeightRanges = [ { score: 7, minimumDensity: 0.890, maximumDensity: 0.910 }, { score: 8, minimumDensity: 0.910, maximumDensity: 0.925 }, { score: 9, minimumDensity: 0.925, maximumDensity: 0.935 }, { score: 10, minimumDensity: 0.935, maximumDensity: 0.950 } ]; const scoreItems = scoreWeightRanges .map(function(range) { const minimumWeight = (range.minimumDensity * volume) / watermelonType.rindCoefficient; const maximumWeight = (range.maximumDensity * volume) / watermelonType.rindCoefficient; return `
${range.score} баллов ${minimumWeight.toFixed(2)}–${maximumWeight.toFixed(2)} кг
`; }) .join(""); hintDiv.style.display = "block"; hintDiv.innerHTML = `
⚖️ Какой вес даст прогноз
${scoreItems}
`; }
